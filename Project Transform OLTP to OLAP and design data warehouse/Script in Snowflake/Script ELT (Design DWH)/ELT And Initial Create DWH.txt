        CREATE OR REPLACE TEMP TABLE stg_dim_creditcard AS
        SELECT 
            CreditCardID,
            CardType,
            CardNumber,
            ExpMonth, 
            ExpYear,
            ModifiedDate,
            IsCurrent,
            IsUpdate,
            (ExpYear * 12 + ExpMonth) - (EXTRACT(YEAR FROM CURRENT_DATE) * 12 + EXTRACT(MONTH FROM CURRENT_DATE)) AS MonthExpiration,
            CASE 
                WHEN (ExpYear * 12 + ExpMonth) >= (EXTRACT(YEAR FROM CURRENT_DATE) * 12 + EXTRACT(MONTH FROM CURRENT_DATE)) 
                    THEN 'Active'
                ELSE 'Expired'
            END AS CardStatus,
            CASE 
                WHEN (ExpYear * 12 + ExpMonth) - (EXTRACT(YEAR FROM CURRENT_DATE) * 12 + EXTRACT(MONTH FROM CURRENT_DATE)) BETWEEN 0 AND 3 
                    THEN 'Active'
                ELSE 'Expired'
            END AS Expiring3Months,
            ModifiedDate AS StartDate,
            SHA2(CONCAT_WS('|', 
                COALESCE(CAST(CreditCardID AS STRING), 'unknown'),
                COALESCE(CardType, 'unknown'),
                COALESCE(CardNumber, 'unknown'),
                COALESCE(CAST(ExpMonth AS STRING), 'unknown'),
                COALESCE(CAST(ExpYear AS STRING), 'unknown')
            ), 256) as HashDiff,
            CASE 
                WHEN IsDelete THEN TRUE 
                ELSE FALSE 
            END AS IsDeleted
        FROM Sales.CreditCard
        WHERE IsInsert = TRUE OR IsUpdate = TRUE OR IsDelete = TRUE;
        
        
        -- STEP 2: UPDATE DÒNG CŨ KHI HASH KHÁC
        UPDATE Sales_Product_DWH.DimCreditCard dwh
        SET 
            IsCurrent = FALSE,  
            EndDate = stg.ModifiedDate
        FROM stg_dim_creditcard stg
        WHERE dwh.CreditCardID = stg.CreditCardID
          AND dwh.IsCurrent = TRUE
          AND dwh.HashDiff != stg.HashDiff
          AND stg.IsCurrent = TRUE;
        
        
        -- STEP 3: INSERT DỮ LIỆU MỚI (KHÁC HASH VÀ KHÔNG TỒN TẠI)
        
            INSERT INTO Sales_Product_DWH.DimCreditCard (

                CardType,
                CardNumber,
                ExpMonth,
                ExpYear,
                MonthExpiration,
                CardStatus,
                Expiring3Months,
                CreditModifiedDate,
                IsCurrent,
                StartDate,
                EndDate,
                HashDiff

                stg.CreditCardID,

                CASE WHEN stg.IsDeleted THEN 'not exist' ELSE stg.CardNumber END,
                stg.ExpMonth,

            prosub.ModifiedDate as SubCateProModifiedDate,

            -- Tính giá trị xếp hạng theo subcategory
            DENSE_RANK() OVER (PARTITION BY prosub.ProductSubcategoryID ORDER BY pro.ListPrice DESC) AS PriceRankInSubcategory,
            
            -- Tính tỷ lệ lợi nhuận
            ROUND((pro.ListPrice - pro.StandardCost) / NULLIF(pro.StandardCost, 0) * 100, 2) AS ProfitMarginPct,
            
            -- Tổng sản phẩm trong category
            COUNT(pro.ProductID) OVER (PARTITION BY procat.ProductCategoryID) AS TotalProInCate,
            
            -- Trung bình giá trong subcategory
            ROUND(AVG(pro.ListPrice) OVER (PARTITION BY prosub.ProductSubcategoryID), 2) AS AvgPriceBySubcategory,
            
            TO_TIMESTAMP('2006-06-23 00:00:00.000', 'YYYY-MM-DD HH24:MI:SS.FF3') AS StartDate,

                COALESCE(TO_VARCHAR(pro.FinishedGoodsFlag), '') || '|' ||

                COALESCE(TO_VARCHAR(pro.SafetyStockLevel), '') || '|' ||

                COALESCE(procat.Name, '')

        LEFT JOIN SALES_PRODUCT_DWH.FactProduct pro 
            ON woou.ProductID = pro.ProductID
        WHERE woou.IsInsert = TRUE OR woou.IsUpdate = TRUE OR woou.IsDelete = TRUE;
        
        -- STEP 1: Update bản ghi cũ khi HashDiff khác
        UPDATE SALES_PRODUCT_DWH.DimWorkOrderRouting dwh
        SET 
            dwh.IsCurrent = FALSE,
            dwh.EndDate = CURRENT_TIMESTAMP()
        FROM stg_dim_workorderrouting stg
        WHERE dwh.WorkOrderID = stg.WorkOrderID
          AND dwh.LocationID = stg.LocationID
          AND dwh.IsCurrent = TRUE
          AND dwh.HashDiff != stg.HashDiff
          AND stg.IsCurrent = TRUE;
        
        -- STEP 2: Insert bản ghi mới (khi HashDiff khác hoặc chưa tồn tại)
        INSERT INTO SALES_PRODUCT_DWH.DimWorkOrderRouting (
            WorkOrderID,
            ProductID,
            ProductKey,
            OperationSequence,
            LocationID,
            LocationName,
            CostRate,
            Availability,
            LocationModifiedDate,
            ScheduledStartDate,
            ScheduledEndDate,
            ActualStartDate,
            ActualEndDate,
            ActualResourceHrs,
            PlannedCost,
            ActualCost,
            RoutingModifiedDate,
            ScrapReasonName,
            ScrapModifiedDate,
            IsCurrent,
            StartDate,
            EndDate,
            HashDiff
        )
        SELECT
            stg.WorkOrderID,
            stg.ProductID,
            stg.ProductKey,
            stg.OperationSequence,
            stg.LocationID,
            stg.LocationName,
            stg.CostRate,
            stg.Availability,
            stg.LocationModifiedDate,
            stg.ScheduledStartDate,
            stg.ScheduledEndDate,
            stg.ActualStartDate,
            stg.ActualEndDate,
            stg.ActualResourceHrs,
            stg.PlannedCost,
            stg.ActualCost,
            stg.RoutingModifiedDate,
            stg.ScrapReasonName,
            stg.ScrapModifiedDate,
            TRUE AS IsCurrent,
            stg.StartDate,
            stg.EndDate,
            stg.HashDiff
        FROM stg_dim_workorderrouting stg
        LEFT JOIN SALES_PRODUCT_DWH.DimWorkOrderRouting dim
            ON stg.WorkOrderID = dim.WorkOrderID
            AND stg.LocationID = dim.LocationID
            AND dim.IsCurrent = TRUE
        WHERE (dim.WorkOrderID IS NULL OR stg.HashDiff != dim.HashDiff)
          AND stg.IsCurrent = TRUE;
        
        -- STEP 3: Handle delete (IsDelete = TRUE)
        UPDATE SALES_PRODUCT_DWH.DimWorkOrderRouting dwh
        SET 
            dwh.IsCurrent = FALSE,
            dwh.EndDate = CURRENT_TIMESTAMP()
        FROM stg_dim_workorderrouting stg
        WHERE dwh.WorkOrderID = stg.WorkOrderID
          AND dwh.LocationID = stg.LocationID
          AND dwh.IsCurrent = TRUE
          AND stg.IsCurrent = FALSE;
          
        RETURN 'ETL Process completed successfully';
END;
$$;

-------------- LOAD FACT TABLE
CREATE OR REPLACE PROCEDURE RUN_LOAD_FACT_DWH_ETL()
RETURNS STRING
LANGUAGE SQL
AS
$$
BEGIN
           --- UPLOAD DATA TABLE DIMDATE DWH
        INSERT INTO SALES_PRODUCT_DWH.DimDate (
            DateKey,
            FullDate,
            Day,
            DayName,
            DayOfWeek,
            IsWeekend,
            WeekOfYear,
            WeekStartDate,
            WeekEndDate,
            Month,
            MonthName,
            MonthYear,
            Quarter,
            QuarterName,
            QuarterYear,
            Year,
            YearHalf,
            YearMonth,
            IsHoliday,
            IsFirstDayOfMonth,
            IsLastDayOfMonth
        )
        
        SELECT
          TO_CHAR(d.FullDate, 'YYYYMMDD') AS DateKey,
          d.FullDate,
          
          DAY(d.FullDate) AS Day,
          DAYNAME(d.FullDate) AS DayName,  
          DAYOFWEEK(d.FullDate) AS DayOfWeek,
          CASE WHEN DAYOFWEEK(d.FullDate) IN (1, 7) THEN TRUE ELSE FALSE END AS IsWeekend,
        
          WEEKOFYEAR(d.FullDate) AS WeekOfYear,
          DATE_TRUNC('WEEK', d.FullDate) AS WeekStartDate,
          DATEADD(DAY, 6, DATE_TRUNC('WEEK', d.FullDate)) AS WeekEndDate,
        
          MONTH(d.FullDate) AS Month,
          TO_CHAR(d.FullDate, 'MMMM') AS MonthName,
          TO_CHAR(d.FullDate, 'YYYY-MM') AS MonthYear,
        
          QUARTER(d.FullDate) AS Quarter,
          CONCAT('Q', QUARTER(d.FullDate)) AS QuarterName,
          CONCAT('Q', QUARTER(d.FullDate), '/', YEAR(d.FullDate)) AS QuarterYear,
        
          YEAR(d.FullDate) AS Year,
          CASE WHEN MONTH(d.FullDate) <= 6 THEN 1 ELSE 2 END AS YearHalf,
          TO_NUMBER(TO_CHAR(d.FullDate, 'YYYYMM')) AS YearMonth,
        
          FALSE AS IsHoliday,
          CASE WHEN DAY(d.FullDate) = 1 THEN TRUE ELSE FALSE END AS IsFirstDayOfMonth,
          CASE WHEN DAY(d.FullDate) = DAY(LAST_DAY(d.FullDate)) THEN TRUE ELSE FALSE END AS IsLastDayOfMonth
        
        FROM (
          SELECT 
            DATEADD(DAY, SEQ4(), '2005-01-01') AS FullDate
          FROM TABLE(GENERATOR(ROWCOUNT => 8036))
        ) d;
        
        UPDATE SALES_PRODUCT_DWH.DimDate
        SET IsHoliday = TRUE
        WHERE 
            (Month = 1 AND Day = 1)      -- New Year
         OR (Month = 4 AND Day = 30)     -- Reunification Day
         OR (Month = 5 AND Day = 1)      -- Labour Day
         OR (Month = 9 AND Day = 2);     -- National Day


         
        --- LOAD FACT
        --- TABLE FACTPURCHASORDERHEADER LOAD 
        ALTER TABLE SALES_PRODUCT_DWH.FactPurchaseOrderHeader
        CLUSTER  BY (OrderDate);
        
        CREATE OR REPLACE TEMP TABLE TempFactPurchaseOrderHeader AS
        WITH dimshipmethod_filtered AS (
          SELECT *, 
                 ROW_NUMBER() OVER (
                     PARTITION BY ShipMethodID
                     ORDER BY StartDate DESC
                 ) AS rn
          FROM SALES_PRODUCT_DWH.DimShipMethod
          WHERE IsCurrent = TRUE 
             OR CURRENT_DATE BETWEEN StartDate AND EndDate
        )
        
        SELECT 
            poh.PurchaseOrderID,
            poh.RevisionNumber,
            poh.Status,
            sm.ShipMethodKey,
            poh.OrderDate,
            poh.ShipDate,
            poh.SubTotal,
            poh.TaxAmt,
            poh.Freight,
            poh.TotalDue,
            poh.ModifiedDate AS PurchaseModifiedDate,
        FROM purchasing.PurchaseOrderHeader AS poh
        LEFT JOIN dimshipmethod_filtered AS sm
          ON poh.ShipMethodID = sm.ShipMethodID
             AND poh.OrderDate BETWEEN sm.StartDate AND sm.EndDate;
        
        INSERT INTO SALES_PRODUCT_DWH.FactPurchaseOrderHeader (
            PurchaseOrderID,
            RevisionNumber,
            Status,
            ShipMethodKey,
            OrderDate,
            ShipDate,
            SubTotal,
            TaxAmt,
            Freight,
            TotalDue,
            PurchaseModifiedDate,
            IsCurrent
        )
        SELECT 
            PurchaseOrderID,
            RevisionNumber,
            Status,
            ShipMethodKey,
            OrderDate,
            ShipDate,
            SubTotal,
            TaxAmt,
            Freight,
            TotalDue,
            PurchaseModifiedDate,
            TRUE AS IsCurrent,
        FROM TempFactPurchaseOrderHeader
        WHERE OrderDate BETWEEN '2011-04-16 00:00:00.000' AND '2014-09-22 00:00:00.000';
        
        
        
        --- TABLE FactSalesOrderDetail LOAD
        ALTER TABLE SALES_PRODUCT_DWH.FactSalesOrderDetail
        CLUSTER  BY (DetailModifiedDate);
        
        CREATE OR REPLACE TEMP TABLE TempFactSalesOrderDetail AS
        SELECT 
            sod.SalesOrderID,
            sod.SalesOrderDetailID,
            sod.CarrierTrackingNumber,
            sod.OrderQty,
            pro.ProductKey,
            so.SpecialOfferKey,
            fsoh.SalesOrderHeaderKey,
            sod.UnitPrice,
            sod.UnitPriceDiscount,
            sod.LineTotal,
            
            -- Window aggregations
            SUM(sod.OrderQty) OVER (PARTITION BY sod.SalesOrderID) AS TotalOrderQtyPerOrder,
            SUM(sod.LineTotal) OVER (PARTITION BY sod.SalesOrderID) AS TotalLineTotalPerOrder,
            RANK() OVER (PARTITION BY sod.SalesOrderID ORDER BY sod.LineTotal DESC) AS ProductRankInOrder,
            sod.ModifiedDate AS DetailModifiedDate,
        
        FROM sales.SalesOrderDetail AS sod
        LEFT JOIN SALES_PRODUCT_DWH.DimSpecialOffer AS so
            ON sod.SpecialOfferID = so.SpecialOfferID
        LEFT JOIN SALES_PRODUCT_DWH.FactProduct AS pro
            ON sod.ProductID = pro.ProductID
        LEFT JOIN SALES_PRODUCT_DWH.FactSalesOrderHeader AS fsoh
            ON sod.SalesOrderID = fsoh.SalesOrderID
        WHERE sod.ModifiedDate BETWEEN '2011-05-31 00:00:00.000' AND '2014-06-30 00:00:00.000';  -- Partition insert range

        INSERT INTO SALES_PRODUCT_DWH.FactSalesOrderDetail (
            SalesOrderID,
            SalesOrderDetailID,
            CarrierTrackingNumber,
            OrderQty,
            ProductKey,
            SpecialOfferKey,
            SalesOrderHeaderKey,
            UnitPrice,
            UnitPriceDiscount,
            LineTotal,
            TotalOrderQtyPerOrder,
            TotalLineTotalPerOrder,
            ProductRankInOrder,
            DetailModifiedDate,
            IsCurrent
        )
        SELECT 
            SalesOrderID,
            SalesOrderDetailID,
            CarrierTrackingNumber,
            OrderQty,
            ProductKey,
            SpecialOfferKey,
            SalesOrderHeaderKey,
            UnitPrice,
            UnitPriceDiscount,
            LineTotal,
            TotalOrderQtyPerOrder,
            TotalLineTotalPerOrder,
            ProductRankInOrder,
            DetailModifiedDate,
            TRUE AS IsCurrent
        FROM TempFactSalesOrderDetail;
        
        
        --- TABLE FactPurchaseOrderDetail LOAD
        -- Step cuối: Insert dữ liệu mới hoặc dữ liệu có thay đổi vào bảng FactPurchaseOrderDetail
        
        ALTER TABLE SALES_PRODUCT_DWH.FactPurchaseOrderDetail
        CLUSTER  BY (DueDate);
        
        CREATE OR REPLACE TEMP TABLE TempFactPurchaseOrderDetail AS
        SELECT 
            pod.PurchaseOrderID,
            pod.PurchaseOrderDetailID,
            pod.DueDate,
            pod.OrderQty,
            poh.PurchaseOrderHeaderKey,
            pro.ProductKey,
            pod.UnitPrice,
            pod.LineTotal,
            pod.ReceivedQty,
            pod.RejectedQty,
            pod.StockedQty,
            pod.ModifiedDate AS OrderDetailModifiedDate
        FROM purchasing.PurchaseOrderDetail AS pod
        LEFT JOIN SALES_PRODUCT_DWH.FactPurchaseOrderHeader AS poh
            ON pod.PurchaseOrderID = poh.PurchaseOrderID
        LEFT JOIN SALES_PRODUCT_DWH.FactProduct AS pro
            ON pod.ProductID = pro.ProductID
        WHERE pod.DueDate BETWEEN '2011-04-30 00:00:00.000' AND '2014-10-22 00:00:00.000';  -- Partition insert range
        
        INSERT INTO SALES_PRODUCT_DWH.FactPurchaseOrderDetail (
            PurchaseOrderID,
            PurchaseOrderDetailID,
            PurchaseOrderHeaderKey,
            ProductKey,
            DueDate,
            OrderQty,
            UnitPrice,
            LineTotal,
            ReceivedQty,
            RejectedQty,
            StockedQty,
            OrderDetailModifiedDate,
            IsCurrent
        )
        SELECT 
            PurchaseOrderID,
            PurchaseOrderDetailID,
            PurchaseOrderHeaderKey,
            ProductKey,
            DueDate,
            OrderQty,
            UnitPrice,
            LineTotal,
            ReceivedQty,
            RejectedQty,
            StockedQty,
            OrderDetailModifiedDate,
            TRUE AS IsCurrent
        FROM TempFactPurchaseOrderDetail;
        
        
        --- Create TEMP TABLE FactSalesOrderHeader (ETL)
        CREATE OR REPLACE TEMP TABLE TempFactSalesOrderHeader AS
        With table_transform as (
        Select SalesOrderID
        , OrderDate
        , DueDate
        , ShipDate
        , ModifiedDate as SalesHeaderModifiedDate
        , CASE WHEN Status = 1 THEN 'InProcess'
        WHEN Status = 2 THEN  'Approved'
        WHEN Status = 3 THEN 'Backordered'
        WHEN Status = 4 THEN 'Rejected'
        WHEN Status = 5 THEN  'Shipped'
        ELSE 'Unknown'
        END AS StatusOder
        , OnlineOrderFlag as OrderOnline
        , PurchaseOrderNumber
        , SalesOrderNumber 
        , AccountNumber
        , CustomerID
        , SalesPersonID
        , TerritoryID
        , CreditCardApprovalCode
        , CurrencyRateID
        , CreditCardID
        , ShipMethodID
        , SubTotal
        , TaxAmt
        , Freight
        , TotalDue
        , SUM(TotalDue) OVER (ORDER BY OrderDate ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS AccumulatingTotalDue
        , COUNT(SalesOrderID) OVER (PARTITION BY SalesPersonID) AS TotalOrdersSales
        , DATEDIFF(DAY, MAX(OrderDate) OVER (PARTITION BY CustomerID), '2014-08-01') AS Recency
        , COUNT(SalesOrderID) OVER (PARTITION BY CustomerID) AS Frequency
        , SUM(TotalDue) OVER (PARTITION BY CustomerID) AS Monetary
        , ROUND(
           TotalDue * 100.0 / SUM(TotalDue) OVER (PARTITION BY CustomerID), 2
        ) || '%' AS CustomerRevenuePctEachDay
        ,  CASE WHEN CreditCardApprovalCode IS NOT NULL THEN 'CreditCard'
            ELSE 'Other'
          END AS PaymentMethod
        , COUNT(*) OVER (PARTITION BY CreditCardApprovalCode) AS TotalOrders
        , COUNT(DISTINCT AccountNumber) OVER (PARTITION BY CreditCardApprovalCode) AS DistinctAccounts
        , MIN(OrderDate) OVER (PARTITION BY CreditCardApprovalCode) AS FirstUseDate
        , MAX(OrderDate) OVER (PARTITION BY CreditCardApprovalCode) AS LastUseDate
        From Sales.SalesOrderHeader),
        
        dimcustomer_filtered AS (
          SELECT *, ROW_NUMBER() OVER (
              PARTITION BY CustomerID
              ORDER BY StartDate DESC
          ) AS rn
          FROM SALES_PRODUCT_DWH.DIMCUSTOMER
          WHERE IsCurrent = TRUE OR CURRENT_DATE BETWEEN StartDate AND EndDate
        ),
        
        dimperson_filtered AS (
          SELECT *, ROW_NUMBER() OVER (
              PARTITION BY BusinessEntityID
              ORDER BY StartDate DESC
          ) AS rn
          FROM SALES_PRODUCT_DWH.DIMSALESPERSON
          WHERE IsCurrent = TRUE OR CURRENT_DATE BETWEEN StartDate AND EndDate
        ),
        
        dimcurrency_filtered AS (
          SELECT *, ROW_NUMBER() OVER (
              PARTITION BY CurrencyRateID
              ORDER BY StartDate DESC
          ) AS rn
          FROM SALES_PRODUCT_DWH.DIMCURRENCY
          WHERE IsCurrent = TRUE OR CURRENT_DATE BETWEEN StartDate AND EndDate
        ),
        
        dimterritory_filtered AS (
          SELECT *, ROW_NUMBER() OVER (
              PARTITION BY TerritoryID
              ORDER BY StartDate DESC
          ) AS rn
          FROM SALES_PRODUCT_DWH.DIMTERRITORY
          WHERE IsCurrent = TRUE OR CURRENT_DATE BETWEEN StartDate AND EndDate
        ),
        
        dimcreditcard_filtered AS (
          SELECT *, ROW_NUMBER() OVER (
              PARTITION BY CreditCardID
              ORDER BY StartDate DESC
          ) AS rn
          FROM SALES_PRODUCT_DWH.DIMCREDITCARD
          WHERE IsCurrent = TRUE OR CURRENT_DATE BETWEEN StartDate AND EndDate
        ),
        
        dimshipmethod_filtered AS (
          SELECT *, ROW_NUMBER() OVER (
              PARTITION BY ShipMethodID
              ORDER BY StartDate DESC
          ) AS rn
          FROM SALES_PRODUCT_DWH.DIMSHIPMETHOD
          WHERE IsCurrent = TRUE OR CURRENT_DATE BETWEEN StartDate AND EndDate
        )        
        
        -- SELECT FINAL
        SELECT ohd.SalesOrderID,
               ohd.OrderDate,
               ohd.DueDate,
               ohd.ShipDate,
               ohd.StatusOder,
               ohd.OrderOnline AS IsOnline,
               ohd.SalespersonID,
               -- Nếu NULL thì gán "Have no Unknown"
               COALESCE(ohd.PurchaseOrderNumber, 'Unknown') AS PurchaseOrderNumber,
               ohd.SalesOrderNumber,
               ohd.AccountNumber AS CusAccount,
               dcus.CustomerKey,
               COALESCE(dper.SalesPersonKey, -1) AS SalesPersonKey,
               CASE WHEN COALESCE(dper.SalesPersonKey, -1) = -1 
                    THEN 'Have No SalesPerson' 
                    ELSE 'Have SalesPerson' 
               END AS SalesPerExplain,
               dter.TerritoryKey,
               COALESCE(dcurr.CurrencyRateKey, -1) AS CurrencyRateKey,
               CASE WHEN COALESCE(dcurr.CurrencyRateKey, -1) = -1 
                    THEN 'Have No CurrencyRate' 
                    ELSE 'Have CurrencyRate' 
              END AS CurrencyExplain,
              COALESCE(dcre.CreditCardKey, -1) AS CreditCardKey,
              CASE WHEN COALESCE(dcre.CreditCardKey, -1) = -1 
                   THEN 'Have No CreditCard' 
                   ELSE 'Have CreditCard' 
              END AS CreditCardExplain,
              ShipMethodKey,
               ddate_1.DateKey AS DateKeyOrder,
               ddate_2.DateKey AS DateKeyDue,
               ddate_3.DateKey AS DateKeyShip,
               -- Nếu NULL thì gán "Unknown"
               COALESCE(ohd.CreditCardApprovalCode, 'Unknown') AS CreditCardApprovalCode,
               ohd.SubTotal,
               ohd.TaxAmt AS Tax,
               ohd.Freight AS ShippingFee,
               ohd.TotalDue AS TotalAmount,
               AccumulatingTotalDue AS CumulativeAmount,
               ohd.TotalOrdersSales AS OrderCount,
               ohd.Recency,
               ohd.Frequency,
               ohd.Monetary,
               PaymentMethod AS PayMethod,
               CASE 
                   WHEN TotalOrders >= 3 AND DistinctAccounts > 1 AND DATEDIFF(DAY, FirstUseDate, LastUseDate) <= 7 
                   THEN 'Fraud Suspected'
                   ELSE 'Normal'
               END AS FraudStatus,
               SalesHeaderModifiedDate,
               TRUE AS IsCurrent
            FROM table_transform AS ohd
            LEFT JOIN dimcustomer_filtered dcus
              ON ohd.CustomerID = dcus.CustomerID
              AND ohd.OrderDate BETWEEN dcus.StartDate AND dcus.EndDate
              AND dcus.rn = 1
            LEFT JOIN dimperson_filtered dper
              ON ohd.SalesPersonID = dper.BusinessEntityID
              AND ohd.OrderDate BETWEEN dper.StartDate AND dper.EndDate
              AND dper.rn = 1
            LEFT JOIN dimcurrency_filtered dcurr
              ON ohd.CurrencyRateID = dcurr.CurrencyRateID
              AND ohd.OrderDate BETWEEN dcurr.StartDate AND dcurr.EndDate
              AND dcurr.rn = 1
            LEFT JOIN dimterritory_filtered dter
              ON ohd.TerritoryID = dter.TerritoryID
              AND ohd.OrderDate BETWEEN dter.StartDate AND dter.EndDate
              AND dter.rn = 1
            LEFT JOIN dimcreditcard_filtered dcre
              ON ohd.CreditCardID = dcre.CreditCardID
              AND ohd.OrderDate BETWEEN dcre.StartDate AND dcre.EndDate
              AND dcre.rn = 1
            Left JOIN dimshipmethod_filtered dsmd
              ON ohd.ShipMethodID = dsmd.ShipMethodID
              AND ohd.OrderDate BETWEEN dsmd.StartDate AND dsmd.EndDate
              AND dsmd.rn = 1 
            LEFT JOIN SALES_PRODUCT_DWH.DIMDATE ddate_1
              ON ohd.OrderDate = ddate_1.FullDate
            LEFT JOIN SALES_PRODUCT_DWH.DIMDATE ddate_2
              ON ohd.DueDate = ddate_2.FullDate
            LEFT JOIN SALES_PRODUCT_DWH.DIMDATE ddate_3
              ON ohd.ShipDate = ddate_3.FullDate;

              -- Insert dữ liệu từ bảng tạm vào bảng FactSalesOrderHeader
        INSERT INTO SALES_PRODUCT_DWH.FactSalesOrderHeader (
            SalesOrderID,
            OrderDate,
            DueDate,
            ShipDate,
            StatusOder,
            IsOnline,
            PurchaseOrderNumber,
            SalesOrderNumber,
            CusAccount,
            SalespersonID,
            CustomerKey,
            SalesPersonKey,
            SalesPerExplain,
            TerritoryKey,
            CurrencyRateKey,
            CurrencyExplain,
            CreditCardKey,
            CreditCardExplain,
            ShipMethodKey,
            DateKeyOrder,
            DateKeyDue,
            DateKeyShip,
            CreditCardApprovalCode,
            SubTotal,
            Tax,
            ShippingFee,
            TotalAmount,
            CumulativeAmount,
            OrderCount,
            Recency,
            Frequency,
            Monetary,
            PayMethod,
            FraudStatus,
            SalesHeaderModifiedDate,
            Iscurrent
        )
        SELECT
            SalesOrderID,
            OrderDate,
            DueDate,
            ShipDate,
            StatusOder,
            IsOnline,
            PurchaseOrderNumber,
            SalesOrderNumber,
            CusAccount,
            SalespersonID,
            CustomerKey,
            SalesPersonKey,
            SalesPerExplain,
            TerritoryKey,
            CurrencyRateKey,
            CurrencyExplain,
            CreditCardKey,
            CreditCardExplain,
            ShipMethodKey,
            DateKeyOrder,
            DateKeyDue,
            DateKeyShip,
            CreditCardApprovalCode,
            SubTotal,
            Tax,
            ShippingFee,
            TotalAmount,
            CumulativeAmount,
            OrderCount,
            Recency,
            Frequency,
            Monetary,
            PayMethod,
            FraudStatus,
            SalesHeaderModifiedDate,
            TRUE AS IsCurrent
        FROM TempFactSalesOrderHeader
        WHERE OrderDate BETWEEN '2011-05-31 00:00:00.000' AND '2014-06-30 00:00:00.000';


        RETURN 'ETL Process completed successfully';
END;
$$;

CALL RUN_LOAD_FACT_DWH_ETL();
